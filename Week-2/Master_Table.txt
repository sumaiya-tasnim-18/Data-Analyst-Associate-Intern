-- 1. Table for Cognito (User) Data
CREATE TABLE Cognito_Raw (
    user_id VARCHAR(255) PRIMARY KEY,
    email VARCHAR(255),
    gender VARCHAR(50),
    UserCreateDate VARCHAR(50),
    UserLastModifiedDate VARCHAR(50),
    birthdate VARCHAR(50),
    city VARCHAR(100),
    zip VARCHAR(20),
    state VARCHAR(100)
);

-- 2. Table for Learner Academic Data
CREATE TABLE Learner_Raw (
    learner_id VARCHAR(255) PRIMARY KEY,
    country VARCHAR(100),
    degree VARCHAR(100),
    institution VARCHAR(255),
    major VARCHAR(255)
);

-- 3. Table for Opportunity Data
CREATE TABLE Opportunity_Raw (
    opportunity_id VARCHAR(255) PRIMARY KEY,
    opportunity_name VARCHAR(255),
    category VARCHAR(100),
    opportunity_code VARCHAR(100),
    tracking_questions TEXT
);

-- 4. Table for Cohort Data
CREATE TABLE Cohort_Raw (
    cohort_id VARCHAR(255),
    cohort_code VARCHAR(100) PRIMARY KEY,
    start_date VARCHAR(50),
    end_date VARCHAR(50),
    size INT
);

-- 5. Table for Learner Enrollment Data (Linking Table)
CREATE TABLE LearnerOpportunity_Raw (
    learner_id VARCHAR(255),
    opportunity_id VARCHAR(255),
    assigned_cohort VARCHAR(100),
    apply_date VARCHAR(50),
    status INT
);

-- Create the final Master_Table by selecting from transformed and joined data
CREATE TABLE Master_Table AS
WITH Cognito_Clean AS (
    -- Clean user data: convert dates and handle NULLs
    SELECT
        user_id,
        email,
        NULLIF(gender, 'NULL') AS gender,
        TO_TIMESTAMP(UserCreateDate, 'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"') AS user_create_date,
        TO_TIMESTAMP(UserLastModifiedDate, 'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"') AS user_last_modified_date,
        TO_DATE(NULLIF(birthdate, 'NULL'), 'MM/DD/YYYY') AS birthdate,
        NULLIF(city, 'NULL') AS city,
        NULLIF(state, 'NULL') AS state
    FROM Cognito_Raw
),
Learner_Clean AS (
    -- Clean learner academic data: remove ID prefix
    SELECT
        REPLACE(learner_id, 'Learner#', '') AS learner_id,
        NULLIF(country, 'NULL') AS country,
        NULLIF(degree, 'NULL') AS degree,
        NULLIF(institution, 'NULL') AS institution,
        NULLIF(major, 'NULL') AS major
    FROM Learner_Raw
),
Opportunity_Clean AS (
    -- Clean opportunity data: remove ID prefix
    SELECT
        REPLACE(opportunity_id, 'Opportunity#', '') AS opportunity_id,
        opportunity_name,
        category,
        opportunity_code
    FROM Opportunity_Raw
),
Cohort_Clean AS (
    -- Clean cohort data: convert Unix timestamps to dates
    SELECT
        cohort_code,
        TO_TIMESTAMP(CAST(start_date AS BIGINT) / 1000) AS start_date,
        TO_TIMESTAMP(CAST(end_date AS BIGINT) / 1000) AS end_date,
        size
    FROM Cohort_Raw
    WHERE start_date <> 'NULL' AND end_date <> 'NULL'
),
LearnerOpportunity_Clean AS (
    -- Clean enrollment data: remove prefixes, map status, and convert date
    SELECT
        REPLACE(learner_id, 'Learner#', '') AS learner_id,
        REPLACE(opportunity_id, 'Opportunity#', '') AS opportunity_id,
        NULLIF(assigned_cohort, 'NULL') AS assigned_cohort,
        TO_TIMESTAMP(apply_date, 'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"') AS apply_date,
        CASE status
            WHEN 1030 THEN 'Applied'
            WHEN 1070 THEN 'Enrolled'
            WHEN 1120 THEN 'Completed'
            WHEN 1130 THEN 'Withdrawn'
            ELSE 'Unknown'
        END AS status
    FROM LearnerOpportunity_Raw
)
-- Final SELECT to join all cleaned CTEs and create the master table
SELECT
    lc.learner_id,
    c.email,
    c.gender,
    c.birthdate,
    EXTRACT(YEAR FROM AGE(NOW(), c.birthdate)) AS age,
    c.city,
    c.state,
    lc.country,
    lc.degree,
    lc.institution,
    lc.major,
    oc.opportunity_id,
    oc.opportunity_name,
    oc.category,
    loc.apply_date,
    loc.status,
    cc.cohort_code,
    cc.start_date,
    cc.end_date,
    cc.size,
    c.user_create_date,
    c.user_last_modified_date
FROM LearnerOpportunity_Clean loc
LEFT JOIN Learner_Clean lc ON loc.learner_id = lc.learner_id
LEFT JOIN Cognito_Clean c ON loc.learner_id = c.user_id
LEFT JOIN Opportunity_Clean oc ON loc.opportunity_id = oc.opportunity_id
LEFT JOIN Cohort_Clean cc ON loc.assigned_cohort = cc.cohort_code
GROUP BY lc.learner_id, c.email, c.gender, c.birthdate, c.city, c.state, lc.country, lc.degree, lc.institution, lc.major, oc.opportunity_id, oc.opportunity_name, oc.category, loc.apply_date, loc.status, cc.cohort_code, cc.start_date, cc.end_date, cc.size, c.user_create_date, c.user_last_modified_date;

-- 1. Record Count Validation
SELECT 'Source Enrollments' AS table_name, COUNT(*) AS record_count FROM LearnerOpportunity_Raw
UNION ALL
SELECT 'Master Table' AS table_name, COUNT(*) AS record_count FROM Master_Table;

-- 2. Duplicate Enrollment Check
SELECT learner_id, opportunity_id, COUNT(*)
FROM Master_Table
GROUP BY learner_id, opportunity_id
HAVING COUNT(*) > 1;

-- 3. Missing Data Review for Key Fields
SELECT
    COUNT(*) AS total_rows,
    COUNT(learner_id) AS count_learner_ids,
    COUNT(opportunity_id) AS count_opportunity_ids,
    COUNT(cohort_code) AS count_cohorts
FROM Master_Table;
